cmake_minimum_required(VERSION 3.25)
project(TargetSymbolConsumer LANGUAGES C CXX)
if(NOT DEFINED CORE_BUILD)
 message(FATAL_ERROR "CORE_BUILD must identify the matching isolated core build")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC AND DMG_ENABLE_ASAN)
 string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
endif()
add_executable(target_symbol "${ROOT}/tests/abi/c_target_symbol.c")
target_include_directories(target_symbol PRIVATE "${ROOT}/include")
target_link_libraries(target_symbol PRIVATE "${CORE_BUILD}/Debug/dexmeguviz_core.lib" "${CORE_BUILD}/Debug/dmg_miniz.lib")
if(MSVC AND DMG_ENABLE_ASAN)
 target_compile_options(target_symbol PRIVATE /fsanitize=address /Zi)
 target_link_options(target_symbol PRIVATE /INCREMENTAL:NO)
endif()
