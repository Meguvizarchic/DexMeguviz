cmake_minimum_required(VERSION 3.25)
project(debug_local_lifecycle_test LANGUAGES C CXX)
set(DMG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DMG_WITH_SQLITE OFF CACHE BOOL "" FORCE)
set(DMG_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. core)
add_executable(debug_local_lifecycle_test main.cpp)
target_compile_features(debug_local_lifecycle_test PRIVATE cxx_std_20)
target_include_directories(debug_local_lifecycle_test PRIVATE ../../src)
target_link_libraries(debug_local_lifecycle_test PRIVATE dexmeguviz_core)
if(MSVC)
  target_compile_definitions(debug_local_lifecycle_test PRIVATE _ITERATOR_DEBUG_LEVEL=0)
  if(DMG_ENABLE_ASAN)
    target_compile_options(debug_local_lifecycle_test PRIVATE /fsanitize=address /Zi)
    target_link_options(debug_local_lifecycle_test PRIVATE /INCREMENTAL:NO)
  endif()
endif()
