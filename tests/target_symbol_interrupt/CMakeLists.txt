cmake_minimum_required(VERSION 3.25)
project(TargetSymbolInterrupt LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DMG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DMG_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}" "${CMAKE_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
target_compile_definitions(dexmeguviz_core PRIVATE DMG_TEST_TARGET_SYMBOL_BARRIER=1)
add_executable(interrupt "${ROOT}/tests/native/target_symbol_interrupt.cpp")
target_compile_features(interrupt PRIVATE cxx_std_20)
target_link_libraries(interrupt PRIVATE dexmeguviz_core)
if(MSVC AND DMG_ENABLE_ASAN)
 target_compile_options(interrupt PRIVATE /fsanitize=address /Zi)
 target_link_options(interrupt PRIVATE /INCREMENTAL:NO)
endif()
