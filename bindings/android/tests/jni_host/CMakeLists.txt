cmake_minimum_required(VERSION 3.25)
project(DexMeguvizJniHost LANGUAGES C CXX)
if(NOT DEFINED JDK_HOME OR NOT EXISTS "${JDK_HOME}/include/jni.h")
  message(FATAL_ERROR "JDK_HOME must point to a local JDK with JNI headers")
endif()
get_filename_component(DEXMEGUVIZ_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(DMG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DMG_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEXMEGUVIZ_ROOT}" "${CMAKE_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
add_library(dexmeguviz_jni_host SHARED "${DEXMEGUVIZ_ROOT}/bindings/android/dexmeguviz-android/src/main/cpp/dexmeguviz_jni.cpp" "${DEXMEGUVIZ_ROOT}/bindings/android/dexmeguviz-android/src/main/cpp/fast_query_jni.cpp")
if(MSVC)
  # The JNI registry has its own mutexes and runs inside the host JVM.
  target_compile_definitions(dexmeguviz_jni_host PRIVATE _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
endif()
target_compile_features(dexmeguviz_jni_host PRIVATE cxx_std_20)
target_include_directories(dexmeguviz_jni_host PRIVATE "${JDK_HOME}/include")
if(WIN32)
  target_include_directories(dexmeguviz_jni_host PRIVATE "${JDK_HOME}/include/win32")
else()
  target_include_directories(dexmeguviz_jni_host PRIVATE "${JDK_HOME}/include/linux")
endif()
target_link_libraries(dexmeguviz_jni_host PRIVATE dexmeguviz_core)
option(DMG_JNI_HOST_PERSISTENT_FACTS "Enable explicit host persistent JNI verification" OFF)
if(DMG_JNI_HOST_PERSISTENT_FACTS)
  target_sources(dexmeguviz_jni_host PRIVATE "${DEXMEGUVIZ_ROOT}/src/cache/persistent.cpp" "${DEXMEGUVIZ_ROOT}/src/cache/persistent_abi.cpp")
  target_include_directories(dexmeguviz_jni_host PRIVATE "${DEXMEGUVIZ_ROOT}/src")
  target_compile_definitions(dexmeguviz_jni_host PRIVATE DMG_ANDROID_PERSISTENT_FACTS=1)
endif()
set_target_properties(dexmeguviz_jni_host PROPERTIES OUTPUT_NAME dexmeguviz)
# Isolated writer checks, not part of the AAR or exported library.
add_executable(dmg_graph_writer_test graph_writer_test.cpp)
target_compile_features(dmg_graph_writer_test PRIVATE cxx_std_20)
target_include_directories(dmg_graph_writer_test PRIVATE "${JDK_HOME}/include")
if(WIN32)
  target_include_directories(dmg_graph_writer_test PRIVATE "${JDK_HOME}/include/win32")
else()
  target_include_directories(dmg_graph_writer_test PRIVATE "${JDK_HOME}/include/linux")
endif()
target_link_libraries(dmg_graph_writer_test PRIVATE dexmeguviz_core)
if(DMG_ENABLE_ASAN)
  if(MSVC)
    target_compile_options(dmg_graph_writer_test PRIVATE /fsanitize=address /Zi)
    target_link_options(dmg_graph_writer_test PRIVATE /INCREMENTAL:NO)
  else()
    target_compile_options(dmg_graph_writer_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dmg_graph_writer_test PRIVATE -fsanitize=address,undefined)
  endif()
endif()

add_executable(dmg_source_bytes_c_test source_bytes_c_test.c)
target_link_libraries(dmg_source_bytes_c_test PRIVATE dexmeguviz_core)

add_executable(dmg_source_symbols_c_test source_symbols_c_test.c)
target_link_libraries(dmg_source_symbols_c_test PRIVATE dexmeguviz_core)

add_executable(dmg_member_symbols_c_test member_symbols_c_test.c)
target_link_libraries(dmg_member_symbols_c_test PRIVATE dexmeguviz_core)
