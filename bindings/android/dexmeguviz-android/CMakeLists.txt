cmake_minimum_required(VERSION 3.25)
project(DexMeguvizAndroid LANGUAGES C CXX)

add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." "${CMAKE_CURRENT_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
set_target_properties(dexmeguviz_core PROPERTIES CXX_VISIBILITY_PRESET default)
add_library(dexmeguviz_android SHARED src/main/cpp/dexmeguviz_jni.cpp src/main/cpp/fast_query_jni.cpp)
target_compile_features(dexmeguviz_android PRIVATE cxx_std_20)
get_filename_component(DMG_INPUT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
target_sources(dexmeguviz_android PRIVATE "${DMG_INPUT_ROOT}/src/fast/native_apk_scan.cpp")
target_include_directories(dexmeguviz_android PRIVATE "${DMG_INPUT_ROOT}/src" "${DMG_INPUT_ROOT}/src/fast")
target_link_libraries(dexmeguviz_android PRIVATE "-Wl,--whole-archive" dexmeguviz_core "-Wl,--no-whole-archive")
target_link_options(dexmeguviz_android PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/exports.map" "-Wl,-z,max-page-size=16384")
set_target_properties(dexmeguviz_android PROPERTIES OUTPUT_NAME dexmeguviz CXX_VISIBILITY_PRESET hidden)
set_property(TARGET dexmeguviz_android APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exports.map")

# Explicit Android-only build variant; default JNI/core and dependency graph remain unchanged.
option(DMG_ANDROID_PERSISTENT_FACTS "Include opt-in Android persistent method-facts bridge" OFF)
if(DMG_ANDROID_PERSISTENT_FACTS)
  if(NOT ANDROID)
    message(FATAL_ERROR "DMG_ANDROID_PERSISTENT_FACTS requires Android; use host test option for JVM checks")
  endif()
  get_filename_component(DMG_PERSISTENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
  target_sources(dexmeguviz_android PRIVATE "${DMG_PERSISTENT_ROOT}/src/cache/persistent.cpp" "${DMG_PERSISTENT_ROOT}/src/cache/persistent_abi.cpp")
  target_include_directories(dexmeguviz_android PRIVATE "${DMG_PERSISTENT_ROOT}/src")
  target_compile_definitions(dexmeguviz_android PRIVATE DMG_ANDROID_PERSISTENT_FACTS=1)
endif()
